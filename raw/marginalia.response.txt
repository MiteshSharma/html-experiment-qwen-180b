    this.el = el;
    this.text = el.dataset.text || el.textContent;
    this.chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz#%&*—+·";
    this.frame = 0;
    this.queue = [];
  }
  Scramble.prototype.start = function(){
    var self = this;
    var old = this.text;
    var len = Math.max(old.length, 1);
    for (var i=0;i<len;i++){
      var start = Math.floor(Math.random()*24);
      var end = start + Math.floor(Math.random()*28) + 8;
      this.queue.push({ char: old[i]||"", start:start, end:end, cur:"" });
    }
    this.raf = null;
    this.update();
  };
  Scramble.prototype.update = function(){
    var out = "", done = 0, self = this;
    for (var i=0;i<this.queue.length;i++){
      var q = this.queue[i];
      if (this.frame >= q.end){ done++; out += q.char; }
      else if (this.frame >= q.start){
        if (!q.cur || Math.random() < 0.3){ q.cur = this.chars[Math.floor(Math.random()*this.chars.length)]; }
        out += q.cur;
      } else {
        out += q.char === " " ? " " : "";
      }
    }
    this.el.textContent = out;
    if (done < this.queue.length){
      this.frame++;
      this.raf = requestAnimationFrame(function(){ self.update(); });
    } else {
      this.el.textContent = this.text;
    }
  };

A few notes on what makes it hold together:

- **Its own identity** — a warm "ink & paper" palette (deep espresso, parchment, gold, vermilion, sage), a Fraunces display / Space Grotesk label / Spectral body pairing, and letterpress/editorial cues (drop caps, marginalia, a Roman-numeral issue mark, hairline rules, section § marks).
- **Characteristic opening** — a full-bleed, breathing (Ken Burns) specimen plate with a masthead that **scramble-decodes** on load, plus a cover masthead strip.
- **Core function wired first** — a working subscribe form with live validation and feedback, plus anchor navigation.
- **Signature motions** — line-mask heading reveals, scramble-decode, Ken Burns breathing, a sticky two-column specimen spread, scattered postcards that straighten on hover, an ink-fill anatomy specimen, a paper-grain overlay, a difference-blend custom cursor with trailing ring, a scroll-progress rule, and a cursor spotlight — all disabled cleanly under `prefers-reduced-motion`.
